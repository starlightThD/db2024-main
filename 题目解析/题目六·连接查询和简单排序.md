## 前置任务

为了方便进行结果对比，题目要求先实现一个简单的单列排序逻辑。
该逻辑需要解析`order by`词法和语法，因此需要修改`lex.l`和`yacc.y`

```
lex.l
"ORDER" { return ORDER; }
"BY" {  return BY;  } # 和GROUP BY复用
"ASC" { return ASC; }
"DESC" { return DESC; } # 和DESC table复用
yacc.y

%type <sv_orderby> orderByClause optOrderByClause
# 为SELECT语句添加ORDER BY子句解析
SELECT selector FROM tableList optWhereClause optGroupByClause optHavingClause optOrderByClause

optOrderByClause:
    ORDER BY orderByClause
    { 
        $$ = $3; 
    }
    |   /* epsilon */ { /* ignore*/ }
    ;

orderByClause:
      col  optAscDesc
    { 
        $$ = std::make_shared<OrderBy>($1, $2);
    }
    ;
optAscDesc:
    ASC          { $$ = OrderBy_ASC;     }
    |  DESC      { $$ = OrderBy_DESC;    }
    |       { $$ = OrderBy_ASC; } # 默认升序
    ;
```

// 排序执行器
class SortExecutor : public AbstractExecutor {
    /* 关键逻辑 */
    void beginTuple() override { 
        cursor_ = 0;
        tuples_.clear();

        prev_->beginTuple();
        for (; !prev_->is_end(); prev_->nextTuple()) {
            auto rec = prev_->Next();
            if (rec != nullptr) {
                tuples_.push_back(std::move(rec));
            }
        }

        std::stable_sort(tuples_.begin(), tuples_.end(),
                         [&](const std::unique_ptr<RmRecord> &lhs, const std::unique_ptr<RmRecord> &rhs) {
                             const char *lhs_raw = lhs->data + sort_col_.offset;
                             const char *rhs_raw = rhs->data + sort_col_.offset;
                             int cmp = compare_raw(lhs_raw, rhs_raw, sort_col_.type, sort_col_.len);
                             return is_desc_ ? (cmp > 0) : (cmp < 0);
                         });
    }
};
